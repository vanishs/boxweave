import json,os,subprocess,sys,time
from pathlib import Path
sys.path.insert(0,"/root/s4c-ready-native/scripts")
from mcp_testkit import LoopbackMCP,MCPError
lab=Path("/var/tmp/s4c-ready-native")
bin=lab/"boxweave"
db=lab/"private/ledger.sqlite"
box="39c49655-f06e-46aa-9137-59e3a17fba51"
url="http://127.0.0.1:18676/mcp"
def ctl(*args):
 r=subprocess.run(["runuser","-u","boxweave","--",str(bin),*args,"-db",str(db),"-project","user-1001"],capture_output=True,text=True,check=True)
 return r.stdout
def issue(ttl):
 s=ctl("token-create","-ttl",ttl)
 return next(l.split("=",1)[1] for l in s.splitlines() if l.startswith("token=")),next(l.split(" ",1)[0].split("=",1)[1] for l in s.splitlines() if l.startswith("id="))
def args(key):
 return dict(box_id=box,idempotency_key="s4c-final-auth-"+key+"-20261120",checks=["running","exec"],timeout_seconds=90)
with LoopbackMCP(url,token=os.environ["BW_TEST_TOKEN"],allow_mutations=True) as inspector:
 for case in ("revoke","expire"):
  secret,tid=issue("1h" if case=="revoke" else "5s")
  try:
   with LoopbackMCP(url,token=secret,allow_mutations=True) as subject:
    r=subject.call("box_wait_ready",args(case)).structured
    assert r["state"]=="OBSERVING"
    oid=r["operation_id"]
    if case=="revoke":ctl("token-revoke","-id",tid)
    else:time.sleep(6)
    for tool,data in (("box_get",dict(box_id=box,detail=True)),("operation_get",dict(operation_id=oid))):
     try:subject.call(tool,data)
     except MCPError as e:
      assert subject.last_status==401 or type(e).__name__=="MCPToolError"
     else:raise AssertionError("invalid credential returned private data")
    until=time.monotonic()+12
    while time.monotonic()<until:
     z=inspector.call("operation_get",dict(operation_id=oid,wait_ms=1000)).structured
     if z["operation"]["state"]=="FAILED":break
    else:raise AssertionError("invalidated observation not settled")
    assert not z["operation"].get("native_operation_id") and not z["readiness"]["ready"]
    print("S4_C_AUTH",case,json.dumps(z,sort_keys=True),flush=True)
  finally:ctl("token-revoke","-id",tid)
 print("S4_C_ACTUAL_REVOKED_AND_EXPIRED_WAIT_NO_DISPATCH_PASS",flush=True)
