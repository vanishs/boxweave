with LoopbackMCP(token=token(receipt),allow_mutations=True,timeout=60) as c:
    status=c.host_status()
    assert status["ready"] and status["commit"]==commit and status["peer_mode"]=="SCOPED_V2"
    def run(action,args,key,box=None):
        request=dict(args,idempotency_key="pr191-wan-lab-"+key,work_ref=work)
        a=c.call(action,request).structured
        oid=a["operation_id"]
        print("ADMITTED",action,key,oid,a["tracking_id"],flush=True)
        v=c.poll_operation(oid,seconds=150)
        assert v["state"]=="SUCCEEDED" and v["tracking_id"]==a["tracking_id"] and v["native_operation_id"]
        assert v["box_uuid"]==v["incarnation"] and (box is None or box==v["box_uuid"])
        expected=(oid,v["tracking_id"],actor,actor,project,action,"SUCCEEDED",v["box_uuid"],v["incarnation"],v["native_operation_id"])
        assert rows("SELECT id,tracking_id,actor_id,token_id,layer,action,state,box_uuid,incarnation,native_operation_id FROM operations WHERE key_digest=?",(hashlib.sha256(request["idempotency_key"].encode()).hexdigest(),))==[expected]
        print("ORIGINAL",key,*expected,flush=True)
        return v
    created=run("box_create",{"name":name,"cpu":1,"memory_mib":512,"disk_gib":4,"image":"local:boxweave-ubuntu-24.04","start":True,"internet":True},"create")
    u=created["box_uuid"]
    def observe(on):
        b=binding(u); n=instance(project,name)
        assert n["config"]["volatile.uuid"]==u and n["expanded_devices"]["eth0"]["security.acls"]==b["acl"] and b["state"]=="PINNED"
        status=c.call("box_get",{"box_id":u}).structured["box"]
        assert status["box_id"]==u and status["internet"] is on
        return b["acl"]
    acl=observe(True)
    def request(key):
        v=run("box_exec",{"box_id":u,"argv":["/bin/sh","-c","LC_ALL=C curl -q -4 --noproxy '*' --proto '=https' --connect-timeout 5 --max-time 8 -fsS -o /dev/null -w '%{http_code} %{time_namelookup} %{time_connect} %{time_appconnect} %{time_total}' https://example.com"]},key,u)
        r=c.call("operation_output_get",{"operation_id":v["operation_id"],"stream":"stdout","limit":4096}).structured
        body=base64.b64decode(r["content_base64"],validate=True).decode("ascii").strip()
        code=r["exit_code"]
        assert rows("SELECT exit_code FROM exec_outputs WHERE operation_id=?",(v["operation_id"],))==[(code,)]
        assert re.fullmatch(r"\d{3}(?: \d{1,3}\.\d{6}){4}",body),body
        print("WAN",key,v["operation_id"],code,body,flush=True)
        return code,body.split()[0]
    # Wait for the ordinary guest address to exist before one actual HTTPS probe.
    ready=run("box_exec",{"box_id":u,"argv":["/bin/sh","-c","for i in $(seq 1 20); do ip -4 route show default | grep -q '^default ' && command -v curl >/dev/null && exit 0; sleep 1; done; exit 1"],"timeout_ms":30000},"ready",u)
    assert rows("SELECT exit_code FROM exec_outputs WHERE operation_id=?",(ready["operation_id"],))==[(0,)]
    first=request("online-before")
    if first[0]!=0 or not re.fullmatch(r"2\d\d",first[1]):
        print("P1B_WAN_LAB_PUBLIC_IPV4_TLS=UNVERIFIED",flush=True)
        sys.exit(2)
    run("box_update",{"box_id":u,"internet":False},"offline-update",u)
    assert observe(False)==acl
    off=request("offline")
    assert off[0]!=0 and off[1]=="000"
    run("box_update",{"box_id":u,"internet":True},"online-update",u)
    assert observe(True)==acl
    after=request("online-after")
    assert after[0]==0 and re.fullmatch(r"2\d\d",after[1])
    assert pid("boxweave.service")==controller
    print("P1B_WAN_LAB_PUBLIC_IPV4_TLS_ON_OFF_ON=PASS",flush=True)
